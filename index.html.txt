    <div class="card"><h2>Affluenza Media per Giorno della Settimana</h2><canvas id="c2"></canvas></div>

<script>
const $ = (id) => document.getElementById(id);
const get = (k, d) => { try { return localStorage.getItem(k) || d; } catch { return d; } };

const cfg = { api: get("api", ""), mq: get("mq", ""), mx: +get("mx", 10) };
$("api").value = cfg.api; $("mq").value = cfg.mq; $("mx").value = cfg.mx;

// Apri le impostazioni se l'API URL non è stato ancora inserito
if (!cfg.api) $("cfg").style.display = "block";

$("save").onclick = () => {
  try {
    localStorage.setItem("api", $("api").value.trim());
    localStorage.setItem("mq", $("mq").value.trim());
    localStorage.setItem("mx", $("mx").value);
  } catch {}
  location.reload();
};

$("cfgb").onclick = () => { const c = $("cfg"); c.style.display = c.style.display === "none" ? "block" : "none"; };
const tab = (n) => { $("live").style.display = n === 1 ? "" : "none"; $("ana").style.display = n === 2 ? "" : "none"; $("t1").className = n === 1 ? "on" : ""; $("t2").className = n === 2 ? "on" : ""; };
$("t1").onclick = () => tab(1); $("t2").onclick = () => tab(2);

// Layout: 2 Piani x 2 Telecamere x 3 Negozi per Telecamera
const CAMS = []; let sn = 1;
for (let f = 1; f <= 2; f++) {
  for (let c = 1; c <= 2; c++) {
    CAMS.push({ id: `F${f}C${c}`, f, shops: [0, 1, 2].map(() => "S" + String(sn++).padStart(2, "0")) });
  }
}
const Z = {}, S = {};

const heat = (v) => `rgba(239,68,68,${(0.06 + 0.8 * Math.min(1, (v || 0) / cfg.mx)).toFixed(2)})`;
const r1 = (v) => Math.round((v || 0) * 10) / 10;
const age = (ts) => { const s = (Date.now() - ts) / 1000; return s < 90 ? Math.round(s) + "s fa" : Math.round(s / 60) + "m fa"; };

function floorSVG(f) {
  let g = "";
  CAMS.filter((c) => c.f === f).forEach((c, k) => {
    const x0 = 10 + k * 315, z = Z[c.id] || {};
    ["left", "center", "right"].forEach((n, i) => {
      const x = x0 + i * 102, v = z[n];
      g += `<rect x="${x}" y="46" width="98" height="104" rx="6" fill="${heat(v || 0)}" stroke="var(--bd)"/>
        <text x="${x + 49}" y="104" text-anchor="middle" font-size="26" font-weight="600" fill="var(--fg)">${v == null ? "–" : r1(v)}</text>
        <text x="${x + 49}" y="140" text-anchor="middle" font-size="11" fill="var(--mut)">${n}</text>`;
    });
    const live = z.ts && Date.now() - z.ts < 180000;
    g += `<circle cx="${x0 + 12}" cy="22" r="8" fill="${live ? "var(--ok)" : "var(--mut)"}"/>
      <text x="${x0 + 28}" y="26" font-size="12" fill="var(--fg)">Cam ${c.id}${z.ts ? " · " + age(z.ts) : ""}</text>`;
    c.shops.forEach((id, i) => {
      const x = x0 + i * 102, s = S[id] || {};
      g += `<rect x="${x}" y="166" width="98" height="84" rx="6" fill="var(--card)" stroke="var(--bd)"/>
        <text x="${x + 8}" y="186" font-size="11" fill="var(--mut)">Negozio ${id}</text>
        <text x="${x + 49}" y="224" text-anchor="middle" font-size="24" font-weight="600" fill="var(--fg)">${s.occ ?? "–"}</text>
        <text x="${x + 49}" y="241" text-anchor="middle" font-size="10" fill="var(--mut)">presenti</text>`;
    });
  });
  return `<svg viewBox="0 0 640 262">${g}</svg>`;
}

function render() {
  $("floors").innerHTML = [1, 2].map((f) => {
    const cs = CAMS.filter((c) => c.f === f);
    const p = cs.reduce((a, c) => a + (Z[c.id] ? (Z[c.id].left || 0) + (Z[c.id].center || 0) + (Z[c.id].right || 0) : 0), 0);
    const sh = cs.reduce((a, c) => a + c.shops.reduce((b, id) => b + ((S[id] || {}).occ || 0), 0), 0);
    return `<div class="card"><h2>Piano ${f} <span class="mut">· ${r1(p)} in galleria · ${sh} nei negozi</span></h2>${floorSVG(f)}</div>`;
  }).join("");
}

const j = (path) => fetch(cfg.api.replace(/\/$/, "") + path).then((r) => { if (!r.ok) throw new Error(r.status); return r.json(); });

async function loadLatest() {
  if (!cfg.api) { $("badge").textContent = "Configura l'API URL in Settings"; return; }
  try {
    const d = await j("/latest");
    if (d.zones) Object.assign(Z, d.zones);
    if (d.shops) Object.assign(S, d.shops);
    render();
    if (!live) $("badge").textContent = "● Polling API attivo";
  } catch (e) { $("badge").textContent = "Errore API: " + e.message; }
}

let live = false;
loadLatest();
setInterval(() => { if (!live) loadLatest(); }, 15000);
setInterval(render, 10000);
render();

// Connessione opzionale tramite MQTT WebSocket
if (cfg.mq) {
  const cl = mqtt.connect(cfg.mq, { clientId: "dash-" + Math.random().toString(16).slice(2, 8) });
  cl.on("connect", () => { live = true; $("badge").textContent = "● MQTT Live"; cl.subscribe("mall/#"); });
  cl.on("close", () => { live = false; $("badge").textContent = "MQTT scollegato, passaggio a Polling API"; });
  cl.on("message", (t, m) => {
    try {
      const d = JSON.parse(m.toString());
      if (t.endsWith("/zones")) Z[d.camera_id] = { ...d.zones, ts: d.ts };
      else if (t.includes("/shop/")) S[d.shop_id] = { occ: d.occupancy, ts: d.ts };
      else return;
      render();
    } catch {}
  });
}

// ---- Sezione Analytics ----
const iso = (d) => new Date(d - d.getTimezoneOffset() * 6e4).toISOString().slice(0, 16);
$("to").value = iso(new Date()); 
$("from").value = iso(new Date(Date.now() - 7 * 864e5)); // Predefinito: ultimi 7 giorni

const charts = {};
function chart(id, type, labels, data, label) {
  if (charts[id]) charts[id].destroy();
  charts[id] = new Chart($(id), {
    type,
    data: {
      labels,
      datasets: [{ label, data, borderColor: "#2563eb", backgroundColor: "rgba(37,99,235,.55)", pointRadius: 0, tension: .25 }]
    },
    options: { plugins: { legend: { display: false } }, responsive: true, maintainAspectRatio: true }
  });
}

const avgBy = (rows, key, n) => {
  const s = Array(n).fill(0), c = Array(n).fill(0);
  rows.forEach(([t, v]) => { const k = key(new Date(t)); s[k] += v; c[k]++; });
  return s.map((v, i) => c[i] ? r1(v / c[i]) : 0);
};

$("load").onclick = async () => {
  if (!cfg.api) return ($("msg").textContent = "Inserisci prima l'API URL nelle impostazioni");
  $("msg").textContent = "Caricamento in corso…";
  
  const from = new Date($("from").value).getTime();
  const to = new Date($("to").value).getTime();
  const q = `&from=${from}&to=${to}&bucket=60`;
  
  try {
    const [z, s] = await Promise.all([j("/history?kind=zones" + q), j("/history?kind=shops" + q)]);
    
    // Aggregazione presenze per timestamp
    const byT = {};
    (z.rows || []).forEach((r) => byT[r.t] = (byT[r.t] || 0) + (r.total || 0));
    
    const ts = Object.keys(byT).map(Number).sort((a, b) => a - b);
    const rows = ts.map((t) => [t, byT[t]]);
    
    // Grafico 1: Presenze nel tempo
    chart("c1", "line", ts.map((t) => new Date(t).toLocaleString([], { day: "2-digit", month: "2-digit", hour: "2-digit" })), ts.map((t) => r1(byT[t])), "Persone");
    
    // Grafico 2: Presenza media per giorno della settimana (Lun-Dom)
    const wd = avgBy(rows, (d) => (d.getDay() + 6) % 7, 7);
    chart("c2", "bar", ["Lun", "Mar", "Mer", "Gio", "Ven", "Sab", "Dom"], wd, "Media Persone");
    
    // Grafico 3: Presenza media per orario
    chart("c3", "bar", [...Array(24).keys()].map((h) => h + ":00"), avgBy(rows, (d) => d.getHours(), 24), "Media Persone");
    
    // Grafico 4: Ingressi medi giornalieri per negozio
    const days = Math.max(1, (to - from) / 864e5);
    const ent = {};
    (s.rows || []).forEach((r) => ent[r.id] = (ent[r.id] || 0) + (r.entries || 0));
    
    const ids = CAMS.flatMap((c) => c.shops);
    chart("c4", "bar", ids, ids.map((id) => r1((ent[id] || 0) / days)), "Ingressi/giorno");
    
    $("msg").textContent = `Dati caricati: ${(z.rows||[]).length} letture zone, ${(s.rows||[]).length} letture negozi.`;
  } catch (e) {
    $("msg").textContent = "Errore durante il caricamento: " + e.message;
  }
};
</script>
